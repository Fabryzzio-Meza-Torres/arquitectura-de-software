def dump(p,x):

def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def minmax(a):
 a=np.asarray(a,float); return np.ones_like(a) if a.max()==a.min() else (a-a.min())/(a.max()-a.min())
def rank(ids,scores):return sorted(range(len(ids)),key=lambda i:(-float(scores[i]),str(ids[i])))
def metrics(ids,rel):
 r=[max(0,rel.get(d,0)) for d in ids]
 ideal=sorted([max(0,v) for v in rel.values()],reverse=True)[:3]
 dcg=lambda a:sum((2**v-1)/math.log2(i+2) for i,v in enumerate(a[:3]))
 den=dcg(ideal)
 return dict(P3=sum(v>0 for v in r[:3])/3, nDCG3=dcg(r)/den if den else 0.,
             MRR10=next((1/(i+1) for i,v in enumerate(r[:10]) if v>0),0.))
def stats(a):
 a=sorted(a);n=len(a)
 return {f'p{q}_ms':a[max(0,math.ceil(q*n/100)-1)] for q in [50,95,99]}
def shard_read(path):
 rows=[]
 with open(path,encoding='utf-8') as f:
  for line in f:
   r=json.loads(line)
   if not isinstance(r.get('_id'),str) or not isinstance(r.get('text'),str):raise ValueError('invalid record')
   # Derived digest only: do not normalize, summarize or overwrite the source text.
   content=(r.get('title','')+' '+r['text']).strip()
   rows.append((r['_id'],hashlib.sha256(content.encode()).hexdigest(),len(content)))
 return rows

class Retriever:
 def __init__(self,docs):
  import torch,bm25s,Stemmer,faiss
  from transformers import AutoModel,AutoTokenizer
  self.torch=torch; self.bm25s=bm25s;self.stem=Stemmer.Stemmer('english')
  self.ids=[str(r['_id']) for r in docs];self.texts=[(r.get('title','')+' '+r['text']).strip() for r in docs]
  t=time.perf_counter()
  toks=bm25s.tokenize(self.texts,stopwords='en',stemmer=self.stem,show_progress=False)
  self.bm=bm25s.BM25(method='lucene',k1=.9,b=.4);self.bm.index(toks,show_progress=False)
  lex_s=time.perf_counter()-t;t=time.perf_counter()
  self.tok=AutoTokenizer.from_pretrained(str(BGE));self.model=AutoModel.from_pretrained(str(BGE)).cuda().half().eval()
  batches=[]
  for start in range(0,len(self.texts),64):
   batches.append(self.encode(self.texts[start:start+64]))
  self.emb=np.concatenate(batches)
  self.index=faiss.IndexFlatIP(self.emb.shape[1]);self.index.add(self.emb);faiss.omp_set_num_threads(1)
  dump(OUT/'indices.json',dict(bm25_seconds=lex_s,semantic_seconds=time.perf_counter()-t,
    bge_revision=BGE.name,documents=len(self.ids),dimension=self.emb.shape[1],pooling='CLS L2 float32, forward fp16',
    bm25=dict(k1=.9,b=.4,stemmer='english',stopwords='en'),index='faiss.IndexFlatIP',original_text_unchanged=True))
  self.bm.save(str(OUT/'bm25_index'));faiss.write_index(self.index,str(OUT/'semantic.faiss'))
  self.query_vectors={}
 def encode(self,texts):
  torch=self.torch
  batch=self.tok(texts,return_tensors='pt',padding=True,truncation=True,max_length=512).to('cuda')
  with torch.inference_mode():v=self.model(**batch).last_hidden_state[:,0];v=torch.nn.functional.normalize(v.float(),p=2,dim=1)
  return v.cpu().numpy().astype('float32')
 def lex(self,q):
  toks=self.bm25s.tokenize(q,stopwords='en',stemmer=self.stem,return_ids=False,show_progress=False)[0]
  return np.asarray(self.bm.get_scores(toks),float) if toks else np.zeros(len(self.ids))
 def sem(self,q):
  v=self.encode(['Represent this sentence for searching relevant passages: '+q])[0]
  values,indices=self.index.search(v[None,:],len(self.ids))
  scores=np.empty(len(self.ids),dtype=float);scores[indices[0]]=values[0]
  return scores
 def search(self,q,parallel=True):
  if parallel:
   with cf.ThreadPoolExecutor(max_workers=2) as pool:
    a=pool.submit(self.lex,q);b=pool.submit(self.sem,q);l,s=a.result(),b.result()
  else:l,s=self.lex(q),self.sem(q)
  li=rank(self.ids,l)[:10];si=rank(self.ids,s)[:10];cand=sorted(set(li)|set(si),key=lambda i:self.ids[i])
  f=.15*minmax(l[cand])+.85*minmax(s[cand])
  return [dict(id=self.ids[i],index=i,lex=float(l[i]),sem=float(s[i]),fused=float(f[j])) for j,i in enumerate(cand)]

class Scorer:
 def __init__(self):
  import torch
  from transformers import AutoTokenizer,AutoModelForCausalLM
  self.torch=torch;self.info=json.loads((ROOT/'modelo_preparado.json').read_text())
  self.tok=AutoTokenizer.from_pretrained(self.info['path']);self.tok.padding_side='left';self.tok.pad_token=self.tok.eos_token
  self.model=AutoModelForCausalLM.from_pretrained(self.info['path'],torch_dtype=torch.bfloat16,attn_implementation='eager').cuda().eval()
  labels=[self.tok.encode(str(i),add_special_tokens=False) for i in range(5)]
  assert all(len(x)==1 for x in labels)
  self.labels=[x[0] for x in labels]
  sig=inspect.signature(self.model.forward).parameters
  self.extra={'num_logits_to_keep':1} if 'num_logits_to_keep' in sig else ({'logits_to_keep':1} if 'logits_to_keep' in sig else {})
 def score(self,pairs,batch):
  result=[];max_batch=0
  for start in range(0,len(pairs),batch):
   part=pairs[start:start+batch];max_batch=max(max_batch,len(part))
   texts=[self.tok.apply_chat_template([{'role':'user','content':PROMPT.format(query=q,doc=t)}],tokenize=False,add_generation_prompt=True) for q,t in part]
   lengths=[len(self.tok.encode(t,add_special_tokens=False)) for t in texts]
   enc=self.tok(texts,return_tensors='pt',padding=True,truncation=True,max_length=1024,add_special_tokens=False).to('cuda')
   with self.torch.inference_mode():logits=self.model(**enc,use_cache=False,**self.extra).logits[:,-1,:].float()
   probs=self.torch.softmax(logits,dim=-1)[:,self.labels];mass=probs.sum(-1)
   norm=probs/mass[:,None].clamp_min(1e-30)
   values=(norm*self.torch.arange(5,device='cuda')).sum(-1)/4
   for i,v in enumerate(values.cpu().tolist()):
    result.append(dict(score=v,label_mass=float(mass[i].cpu()),probabilities=norm[i].cpu().tolist(),
                       truncated=lengths[i]>1024,input_tokens=min(lengths[i],1024),degraded=float(mass[i].cpu())<.5))
  return result,max_batch
